EXPLAIN ANALYZE
 
 WITH 
 pos AS (
     SELECT
         snapshot_ts,
         vehicle_id,
         lat,
         lon,
         st_point(lat, lon) AS geom4326,
         st_transform(
             st_point(lat, lon),
             'EPSG:4326',
             'EPSG:3301'
         ) AS geom3301
     FROM db.standardized.route8_positions
     WHERE fleet_number = 35
 ),
 
 stop AS (
     SELECT
         stop_lat,
         stop_lon,
         st_point(stop_lat, stop_lon) AS geom4326,
         st_transform(
             st_point(stop_lat, stop_lon),
             'EPSG:4326',
             'EPSG:3301'
         ) AS geom3301
     FROM db.standardized.gtfs_stops
     WHERE stop_id = 822
 )
 
 SELECT 
     p.snapshot_ts,
     p.vehicle_id,
     p.lat,
     p.lon,
 
     -- METHOD 1: WGS84 flat-Earth with meridian correction
     sqrt(
         pow((p.lat - s.stop_lat) * (
             111132.954 
             - 559.822 * cos(radians(2 * (p.lat + s.stop_lat) / 2))
             + 1.175  * cos(radians(4 * (p.lat + s.stop_lat) / 2))
         ), 2)
       + pow((p.lon - s.stop_lon) * 111319.5 * cos(radians((p.lat + s.stop_lat) / 2)), 2)
     ) AS eucl_wgs84_manual_m,
 
     -- METHOD 2: EPSG:3301 (Estonia LCC)
     st_distance(
         p.geom3301,
         s.geom3301
     ) AS projected_distance_m,
 
     -- METHOD 3: Haversine (manual)
     (2 * 6370986 * asin(sqrt(
         pow(sin(radians((p.lat - s.stop_lat) / 2)), 2)
       + cos(radians(s.stop_lat)) * cos(radians(p.lat))
       * pow(sin(radians((p.lon - s.stop_lon) / 2)), 2)
     ))) AS haversine_manual_m,
 
     -- METHOD 3b: Built-in Haversine
      st_distance_sphere(
         p.geom4326,
         s.geom4326
     ) AS sphere_distance_m,
 
     -- METHOD 4: Spheroid
     st_distance_spheroid(
         p.geom4326,
         s.geom4326
     ) AS spheroid_distance_m
 
 FROM pos p
 CROSS JOIN stop s
 ORDER BY p.snapshot_ts, p.vehicle_id
